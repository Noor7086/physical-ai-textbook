# Building ROS 2 Python Packages

This chapter covers how to create, structure, and build ROS 2 packages using Python.

## Package Structure

A ROS 2 Python package has the following structure:

```
my_robot_pkg/
├── my_robot_pkg/           # Python module
│   ├── __init__.py
│   ├── robot_node.py
│   └── utils.py
├── resource/
│   └── my_robot_pkg        # Marker file
├── test/
│   └── test_robot.py
├── package.xml             # Package metadata
├── setup.py                # Python setup
└── setup.cfg               # Entry points config
```

## Creating a Package

```bash
# Navigate to your workspace src folder
cd ~/ros2_ws/src

# Create a new package
ros2 pkg create --build-type ament_python my_humanoid_pkg \
    --dependencies rclpy std_msgs geometry_msgs sensor_msgs
```

## package.xml

The package manifest defines metadata and dependencies:

```xml
<?xml version="1.0"?>
<?xml-model href="http://download.ros.org/schema/package_format3.xsd" schematypens="http://www.w3.org/2001/XMLSchema"?>
<package format="3">
  <name>my_humanoid_pkg</name>
  <version>0.1.0</version>
  <description>Humanoid robot control package</description>
  
  <maintainer email="developer@example.com">Developer</maintainer>
  <license>Apache-2.0</license>

  <!-- Build dependencies -->
  <buildtool_depend>ament_python</buildtool_depend>
  
  <!-- Runtime dependencies -->
  <exec_depend>rclpy</exec_depend>
  <exec_depend>std_msgs</exec_depend>
  <exec_depend>geometry_msgs</exec_depend>
  <exec_depend>sensor_msgs</exec_depend>
  
  <!-- Test dependencies -->
  <test_depend>ament_copyright</test_depend>
  <test_depend>ament_flake8</test_depend>
  <test_depend>ament_pep257</test_depend>
  <test_depend>python3-pytest</test_depend>

  <export>
    <build_type>ament_python</build_type>
  </export>
</package>
```

## setup.py

```python
from setuptools import find_packages, setup

package_name = 'my_humanoid_pkg'

setup(
    name=package_name,
    version='0.1.0',
    packages=find_packages(exclude=['test']),
    data_files=[
        ('share/ament_index/resource_index/packages',
            ['resource/' + package_name]),
        ('share/' + package_name, ['package.xml']),
        # Include launch files
        ('share/' + package_name + '/launch', 
            ['launch/robot_launch.py']),
        # Include config files
        ('share/' + package_name + '/config',
            ['config/robot_params.yaml']),
    ],
    install_requires=['setuptools'],
    zip_safe=True,
    maintainer='Developer',
    maintainer_email='developer@example.com',
    description='Humanoid robot control package',
    license='Apache-2.0',
    tests_require=['pytest'],
    entry_points={
        'console_scripts': [
            'robot_controller = my_humanoid_pkg.robot_controller:main',
            'sensor_processor = my_humanoid_pkg.sensor_processor:main',
            'motion_planner = my_humanoid_pkg.motion_planner:main',
        ],
    },
)
```

## Example: Complete Robot Controller

### robot_controller.py

```python
#!/usr/bin/env python3
"""
Robot Controller Node
Coordinates humanoid robot movements
"""

import rclpy
from rclpy.node import Node
from rclpy.qos import QoSProfile, ReliabilityPolicy

from geometry_msgs.msg import Twist, Pose
from sensor_msgs.msg import JointState, Imu
from std_msgs.msg import Float64MultiArray


class RobotController(Node):
    """Main controller for humanoid robot."""
    
    def __init__(self):
        super().__init__('robot_controller')
        
        # Declare parameters
        self.declare_parameter('control_rate', 100.0)
        self.declare_parameter('max_velocity', 1.0)
        
        # Get parameters
        self.control_rate = self.get_parameter('control_rate').value
        self.max_velocity = self.get_parameter('max_velocity').value
        
        # QoS for sensor data
        sensor_qos = QoSProfile(
            reliability=ReliabilityPolicy.BEST_EFFORT,
            depth=1
        )
        
        # Publishers
        self.joint_cmd_pub = self.create_publisher(
            Float64MultiArray,
            '/joint_commands',
            10
        )
        
        # Subscribers
        self.joint_state_sub = self.create_subscription(
            JointState,
            '/joint_states',
            self.joint_state_callback,
            sensor_qos
        )
        
        self.imu_sub = self.create_subscription(
            Imu,
            '/imu/data',
            self.imu_callback,
            sensor_qos
        )
        
        self.cmd_vel_sub = self.create_subscription(
            Twist,
            '/cmd_vel',
            self.cmd_vel_callback,
            10
        )
        
        # State variables
        self.current_joint_positions = []
        self.current_orientation = None
        self.target_velocity = Twist()
        
        # Control loop timer
        period = 1.0 / self.control_rate
        self.control_timer = self.create_timer(period, self.control_loop)
        
        self.get_logger().info(
            f'Robot Controller started at {self.control_rate} Hz'
        )
    
    def joint_state_callback(self, msg: JointState):
        """Process joint state feedback."""
        self.current_joint_positions = list(msg.position)
    
    def imu_callback(self, msg: Imu):
        """Process IMU data for balance."""
        self.current_orientation = msg.orientation
    
    def cmd_vel_callback(self, msg: Twist):
        """Receive velocity commands."""
        # Clamp to max velocity
        self.target_velocity.linear.x = max(
            -self.max_velocity,
            min(self.max_velocity, msg.linear.x)
        )
        self.target_velocity.angular.z = msg.angular.z
    
    def control_loop(self):
        """Main control loop - runs at control_rate Hz."""
        if not self.current_joint_positions:
            return
        
        # Compute desired joint commands based on target velocity
        joint_commands = self.compute_joint_commands()
        
        # Publish commands
        msg = Float64MultiArray()
        msg.data = joint_commands
        self.joint_cmd_pub.publish(msg)
    
    def compute_joint_commands(self):
        """Compute joint commands from velocity target."""
        # Simplified inverse kinematics
        # In practice, this would be much more complex
        
        linear_vel = self.target_velocity.linear.x
        angular_vel = self.target_velocity.angular.z
        
        # Example: 6 joints (simplified humanoid leg)
        commands = [
            0.0,                    # Hip yaw
            0.1 * linear_vel,       # Hip pitch
            -0.2 * linear_vel,      # Knee
            0.1 * linear_vel,       # Ankle pitch
            angular_vel * 0.1,      # Ankle roll
            0.0                     # Toe
        ]
        
        return commands


def main(args=None):
    rclpy.init(args=args)
    controller = RobotController()
    
    try:
        rclpy.spin(controller)
    except KeyboardInterrupt:
        pass
    finally:
        controller.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

## Building and Running

```bash
# Build the workspace
cd ~/ros2_ws
colcon build --packages-select my_humanoid_pkg

# Source the workspace
source install/setup.bash

# Run a node
ros2 run my_humanoid_pkg robot_controller

# Run with parameters
ros2 run my_humanoid_pkg robot_controller --ros-args \
    -p control_rate:=200.0 \
    -p max_velocity:=0.5
```

## Using Parameters

### Parameter Declaration

```python
class ParameterizedNode(Node):
    def __init__(self):
        super().__init__('parameterized_node')
        
        # Declare with default values
        self.declare_parameter('robot_name', 'humanoid_1')
        self.declare_parameter('joint_count', 28)
        self.declare_parameter('control_gains', [1.0, 0.1, 0.01])
        
        # Get values
        self.robot_name = self.get_parameter('robot_name').value
        self.joint_count = self.get_parameter('joint_count').value
        self.gains = self.get_parameter('control_gains').value
        
        # Parameter change callback
        self.add_on_set_parameters_callback(self.parameter_callback)
    
    def parameter_callback(self, params):
        for param in params:
            self.get_logger().info(f'Parameter {param.name} changed to {param.value}')
        return SetParametersResult(successful=True)
```


## Summary

| File | Purpose |
|------|---------|
| `package.xml` | Package metadata and dependencies |
| `setup.py` | Python package configuration |
| `setup.cfg` | Entry points for executables |
| `*.py` | Node implementations |
| `config/*.yaml` | Parameter files |
