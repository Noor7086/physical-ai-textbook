# URDF for Humanoid Robots

**URDF** (Unified Robot Description Format) is an XML format for describing a robot's physical structure. This chapter covers how to create URDF files for humanoid robots.

## URDF Basics

A URDF file defines:
- **Links** - Rigid body parts (torso, limbs, head)
- **Joints** - Connections between links (revolute, prismatic, fixed)
- **Visual** - How the robot looks
- **Collision** - Simplified geometry for physics
- **Inertial** - Mass properties for dynamics

## Basic Structure

```xml
<?xml version="1.0"?>
<robot name="humanoid">
  
  <!-- Base link (torso) -->
  <link name="base_link">
    <visual>
      <geometry>
        <box size="0.3 0.2 0.5"/>
      </geometry>
      <material name="blue">
        <color rgba="0 0 0.8 1"/>
      </material>
    </visual>
    <collision>
      <geometry>
        <box size="0.3 0.2 0.5"/>
      </geometry>
    </collision>
    <inertial>
      <mass value="10.0"/>
      <inertia ixx="0.1" ixy="0" ixz="0" 
               iyy="0.1" iyz="0" izz="0.1"/>
    </inertial>
  </link>
  
  <!-- Hip joint -->
  <joint name="hip_joint" type="revolute">
    <parent link="base_link"/>
    <child link="upper_leg"/>
    <origin xyz="0 -0.1 -0.25" rpy="0 0 0"/>
    <axis xyz="1 0 0"/>
    <limit lower="-1.57" upper="1.57" effort="100" velocity="2.0"/>
  </joint>
  
  <!-- Upper leg link -->
  <link name="upper_leg">
    <visual>
      <origin xyz="0 0 -0.2"/>
      <geometry>
        <cylinder radius="0.05" length="0.4"/>
      </geometry>
    </visual>
    <collision>
      <origin xyz="0 0 -0.2"/>
      <geometry>
        <cylinder radius="0.05" length="0.4"/>
      </geometry>
    </collision>
    <inertial>
      <mass value="3.0"/>
      <origin xyz="0 0 -0.2"/>
      <inertia ixx="0.05" ixy="0" ixz="0" 
               iyy="0.05" iyz="0" izz="0.01"/>
    </inertial>
  </link>

</robot>
```

## Joint Types

| Type | DOF | Description | Use Case |
|------|-----|-------------|----------|
| `revolute` | 1 | Rotation with limits | Most joints |
| `continuous` | 1 | Unlimited rotation | Wheels |
| `prismatic` | 1 | Linear motion | Telescoping |
| `fixed` | 0 | No motion | Welded parts |
| `floating` | 6 | Free in space | Mobile base |
| `planar` | 3 | Motion in plane | Special cases |

## Complete Humanoid Leg URDF

```xml
<?xml version="1.0"?>
<robot name="humanoid_leg" xmlns:xacro="http://www.ros.org/wiki/xacro">

  <!-- Materials -->
  <material name="grey">
    <color rgba="0.5 0.5 0.5 1"/>
  </material>
  <material name="dark_grey">
    <color rgba="0.3 0.3 0.3 1"/>
  </material>

  <!-- Properties -->
  <xacro:property name="leg_length" value="0.4"/>
  <xacro:property name="leg_radius" value="0.04"/>
  <xacro:property name="foot_length" value="0.15"/>
  
  <!-- Pelvis (base) -->
  <link name="pelvis">
    <visual>
      <geometry>
        <box size="0.2 0.3 0.1"/>
      </geometry>
      <material name="grey"/>
    </visual>
    <collision>
      <geometry>
        <box size="0.2 0.3 0.1"/>
      </geometry>
    </collision>
    <inertial>
      <mass value="5.0"/>
      <inertia ixx="0.04" ixy="0" ixz="0" 
               iyy="0.02" iyz="0" izz="0.04"/>
    </inertial>
  </link>

  <!-- Right Hip Yaw Joint -->
  <joint name="r_hip_yaw" type="revolute">
    <parent link="pelvis"/>
    <child link="r_hip_yaw_link"/>
    <origin xyz="0 -0.1 -0.05" rpy="0 0 0"/>
    <axis xyz="0 0 1"/>
    <limit lower="-0.5" upper="0.5" effort="50" velocity="3.0"/>
    <dynamics damping="0.1" friction="0.1"/>
  </joint>

  <link name="r_hip_yaw_link">
    <visual>
      <geometry>
        <sphere radius="0.03"/>
      </geometry>
      <material name="dark_grey"/>
    </visual>
    <collision>
      <geometry>
        <sphere radius="0.03"/>
      </geometry>
    </collision>
    <inertial>
      <mass value="0.5"/>
      <inertia ixx="0.001" ixy="0" ixz="0" 
               iyy="0.001" iyz="0" izz="0.001"/>
    </inertial>
  </link>

  <!-- Right Hip Roll Joint -->
  <joint name="r_hip_roll" type="revolute">
    <parent link="r_hip_yaw_link"/>
    <child link="r_hip_roll_link"/>
    <origin xyz="0 0 0" rpy="0 0 0"/>
    <axis xyz="1 0 0"/>
    <limit lower="-0.5" upper="0.5" effort="50" velocity="3.0"/>
  </joint>

  <link name="r_hip_roll_link">
    <visual>
      <geometry>
        <sphere radius="0.03"/>
      </geometry>
      <material name="dark_grey"/>
    </visual>
    <inertial>
      <mass value="0.5"/>
      <inertia ixx="0.001" ixy="0" ixz="0" 
               iyy="0.001" iyz="0" izz="0.001"/>
    </inertial>
  </link>

  <!-- Right Hip Pitch Joint -->
  <joint name="r_hip_pitch" type="revolute">
    <parent link="r_hip_roll_link"/>
    <child link="r_upper_leg"/>
    <origin xyz="0 0 0" rpy="0 0 0"/>
    <axis xyz="0 1 0"/>
    <limit lower="-1.5" upper="0.5" effort="100" velocity="3.0"/>
  </joint>

  <!-- Right Upper Leg -->
  <link name="r_upper_leg">
    <visual>
      <origin xyz="0 0 -0.2"/>
      <geometry>
        <cylinder radius="${leg_radius}" length="${leg_length}"/>
      </geometry>
      <material name="grey"/>
    </visual>
    <collision>
      <origin xyz="0 0 -0.2"/>
      <geometry>
        <cylinder radius="${leg_radius}" length="${leg_length}"/>
      </geometry>
    </collision>
    <inertial>
      <mass value="3.0"/>
      <origin xyz="0 0 -0.2"/>
      <inertia ixx="0.04" ixy="0" ixz="0" 
               iyy="0.04" iyz="0" izz="0.005"/>
    </inertial>
  </link>

  <!-- Right Knee Joint -->
  <joint name="r_knee" type="revolute">
    <parent link="r_upper_leg"/>
    <child link="r_lower_leg"/>
    <origin xyz="0 0 -0.4" rpy="0 0 0"/>
    <axis xyz="0 1 0"/>
    <limit lower="0" upper="2.5" effort="100" velocity="3.0"/>
  </joint>

  <!-- Right Lower Leg -->
  <link name="r_lower_leg">
    <visual>
      <origin xyz="0 0 -0.2"/>
      <geometry>
        <cylinder radius="${leg_radius}" length="${leg_length}"/>
      </geometry>
      <material name="grey"/>
    </visual>
    <collision>
      <origin xyz="0 0 -0.2"/>
      <geometry>
        <cylinder radius="${leg_radius}" length="${leg_length}"/>
      </geometry>
    </collision>
    <inertial>
      <mass value="2.5"/>
      <origin xyz="0 0 -0.2"/>
      <inertia ixx="0.03" ixy="0" ixz="0" 
               iyy="0.03" iyz="0" izz="0.004"/>
    </inertial>
  </link>

  <!-- Right Ankle Pitch Joint -->
  <joint name="r_ankle_pitch" type="revolute">
    <parent link="r_lower_leg"/>
    <child link="r_ankle_link"/>
    <origin xyz="0 0 -0.4" rpy="0 0 0"/>
    <axis xyz="0 1 0"/>
    <limit lower="-0.8" upper="0.8" effort="50" velocity="3.0"/>
  </joint>

  <link name="r_ankle_link">
    <inertial>
      <mass value="0.3"/>
      <inertia ixx="0.0005" ixy="0" ixz="0" 
               iyy="0.0005" iyz="0" izz="0.0005"/>
    </inertial>
  </link>

  <!-- Right Ankle Roll Joint -->
  <joint name="r_ankle_roll" type="revolute">
    <parent link="r_ankle_link"/>
    <child link="r_foot"/>
    <origin xyz="0 0 0" rpy="0 0 0"/>
    <axis xyz="1 0 0"/>
    <limit lower="-0.5" upper="0.5" effort="50" velocity="3.0"/>
  </joint>

  <!-- Right Foot -->
  <link name="r_foot">
    <visual>
      <origin xyz="0.03 0 -0.02"/>
      <geometry>
        <box size="${foot_length} 0.08 0.04"/>
      </geometry>
      <material name="dark_grey"/>
    </visual>
    <collision>
      <origin xyz="0.03 0 -0.02"/>
      <geometry>
        <box size="${foot_length} 0.08 0.04"/>
      </geometry>
    </collision>
    <inertial>
      <mass value="1.0"/>
      <origin xyz="0.03 0 -0.02"/>
      <inertia ixx="0.002" ixy="0" ixz="0" 
               iyy="0.004" iyz="0" izz="0.004"/>
    </inertial>
  </link>

</robot>
```

## Using Xacro

**Xacro** (XML Macros) makes URDF files more maintainable:

```xml
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro" name="humanoid">

  <!-- Properties -->
  <xacro:property name="leg_mass" value="3.0"/>
  <xacro:property name="leg_length" value="0.4"/>
  
  <!-- Macro for leg segment -->
  <xacro:macro name="leg_segment" params="name parent length mass">
    <joint name="${name}_joint" type="revolute">
      <parent link="${parent}"/>
      <child link="${name}"/>
      <axis xyz="0 1 0"/>
      <limit lower="-1.57" upper="1.57" effort="100" velocity="3.0"/>
    </joint>
    
    <link name="${name}">
      <visual>
        <origin xyz="0 0 ${-length/2}"/>
        <geometry>
          <cylinder radius="0.04" length="${length}"/>
        </geometry>
      </visual>
      <inertial>
        <mass value="${mass}"/>
        <origin xyz="0 0 ${-length/2}"/>
      </inertial>
    </link>
  </xacro:macro>
  
  <!-- Use macro -->
  <xacro:leg_segment name="upper_leg" parent="pelvis" 
                     length="${leg_length}" mass="${leg_mass}"/>
  <xacro:leg_segment name="lower_leg" parent="upper_leg" 
                     length="${leg_length}" mass="${leg_mass}"/>

</robot>
```

## Summary

| Element | Purpose |
|---------|---------|
| `<link>` | Define rigid body with visual, collision, inertial |
| `<joint>` | Connect links with motion constraints |
| `<visual>` | Appearance for visualization |
| `<collision>` | Simplified geometry for physics |
| `<inertial>` | Mass and inertia for dynamics |
| Xacro | Macros for maintainable URDFs |
